```javascript
import mongoose from 'mongoose';

const specialtySchema = new mongoose.Schema({
  name: {
    type: String,
    required: true,
    unique: true,
    trim: true,
    minlength: 2,
    maxlength: 50
  },
  category: {
    type: String,
    enum: ['coupe', 'coloration', 'soin', 'extension', 'autre'],
    required: true
  },
  aliases: [{
    type: String,
    trim: true,
    maxlength: 50
  }],
  description: {
    type: String,
    maxlength: 500
  },
  isActive: {
    type: Boolean,
    default: true
  },
  usageCount: {
    type: Number,
    default: 0,
    min: 0
  },
  createdAt: {
    type: Date,
    default: Date.now
  },
  updatedAt: {
    type: Date,
    default: Date.now
  }
}, {
  timestamps: true
});

// Index pour performance
specialtySchema.index({ name: 'text', aliases: 'text' });
specialtySchema.index({ category: 1, usageCount: -1 });
specialtySchema.index({ isActive: 1, category: 1 });

// Méthodes du modèle
specialtySchema.methods.incrementUsage = async function() {
  this.usageCount += 1;
  await this.save();
  return this;
};

specialtySchema.methods.decrementUsage = async function() {
  this.usageCount = Math.max(0, this.usageCount - 1);
  await this.save();
  return this;
};

// Validation personnalisée
specialtySchema.pre('save', function(next) {
  this.updatedAt = Date.now();
  next();
});

const Specialty = mongoose.model('Specialty', specialtySchema);
export default Specialty;
```

### **JOUR 3-4: CRÉATION DE L'API DES SPÉCIALITÉS**

#### **2.1 Créer le fichier `back/routes/specialties.js`**
```javascript
import express from 'express';
import { authenticateToken, isCoiffeur } from '../middleware/auth.js';
import Specialty from '../models/Specialty.js';

const router = express.Router();

// GET /api/specialties - Récupérer toutes les spécialités actives
router.get('/', async (req, res) => {
  try {
    const specialties = await Specialty.find({ isActive: true })
      .sort({ category: 1, name: 1 });
    res.json(specialties);
  } catch (error) {
    res.status(500).json({ error: 'Erreur serveur' });
  }
});

// GET /api/specialties/category/:category - Par catégorie
router.get('/category/:category', async (req, res) => {
  try {
    const { category } = req.params;
    const specialties = await Specialty.find({ 
      category, 
      isActive: true 
    }).sort({ name: 1 });
    res.json(specialties);
  } catch (error) {
    res.status(500).json({ error: 'Erreur serveur' });
  }
});

// POST /api/specialties/search - Recherche intelligente
router.post('/search', async (req, res) => {
  try {
    const { query, limit = 10 } = req.body;
    
    if (!query || query.trim().length < 2) {
      return res.status(400).json({ error: 'Requête trop courte' });
    }
    
    // 1. Recherche exacte
    let exact = await Specialty.findOne({ 
      name: { $regex: `^${query.trim()}$`, $i: true },
      isActive: true 
    });
    
    if (exact) {
      return res.json({ exact, suggestions: [] });
    }
    
    // 2. Recherche par alias
    let byAlias = await Specialty.find({ 
      aliases: { $regex: query.trim(), $i: true },
      isActive: true 
    });
    
    // 3. Recherche floue
    let fuzzy = await Specialty.find({
      $or: [
        { name: { $regex: query.trim(), $i: true } },
        { aliases: { $regex: query.trim(), $i: true } }
      ],
      isActive: true
    }).sort({ usageCount: -1 }).limit(limit);
    
    // Éviter les doublons
    const allResults = [...byAlias, ...fuzzy];
    const uniqueResults = allResults.filter((item, index, self) => 
      index === self.findIndex(t => t._id.toString() === item._id.toString())
    );
    
    res.json({ 
      exact: null, 
      suggestions: uniqueResults.slice(0, limit) 
    });
    
  } catch (error) {
    console.error('Erreur recherche spécialités:', error);
    res.status(500).json({ error: 'Erreur serveur' });
  }
});

// POST /api/specialties - Créer une nouvelle spécialité (admin uniquement)
router.post('/', authenticateToken, isCoiffeur, async (req, res) => {
  try {
    const { name, category, aliases, description } = req.body;
    
    // Vérifier si la spécialité existe déjà
    const existing = await Specialty.findOne({ 
      name: { $regex: `^${name}$`, $i: true } 
    });
    
    if (existing) {
      return res.status(400).json({ 
        error: 'Cette spécialité existe déjà',
        existingSpecialty: existing 
      });
    }
    
    const specialty = new Specialty({
      name: name.trim(),
      category,
      aliases: aliases || [],
      description
    });
    
    await specialty.save();
    res.status(201).json(specialty);
    
  } catch (error) {
    console.error('Erreur création spécialité:', error);
    res.status(500).json({ error: 'Erreur serveur' });
  }
});

export default router;
```

### **JOUR 5-6: MIGRATION DES DONNÉES EXISTANTES**

#### **3.1 Créer le script de migration**
```javascript
// back/scripts/migrateSpecialties.js
import mongoose from 'mongoose';
import User from '../models/User.js';
import Specialty from '../models/Specialty.js';
import { mongoURI } from '../config/mongoURI.js';

const migrateSpecialties = async () => {
  try {
    await mongoose.connect(mongoURI);
    console.log('✅ Connecté à MongoDB');
    
    // 1. Récupérer toutes les spécialités existantes
    const users = await User.find({ 
      role: 'coiffeur',
      specialities: { $exists: true, $ne: [] }
    });
    
    console.log(`📊 ${users.length} coiffeurs avec spécialités trouvés`);
    
    // 2. Créer les spécialités uniques
    const allSpecialties = new Set();
    users.forEach(user => {
      user.specialities.forEach(spec => {
        if (spec && spec.trim()) {
          allSpecialties.add(spec.trim().toLowerCase());
        }
      });
    });
    
    console.log(`🔍 ${allSpecialties.size} spécialités uniques identifiées`);
    
    // 3. Créer les documents Specialty
    const specialtyDocs = [];
    for (const specName of allSpecialties) {
      // Déterminer la catégorie
      let category = 'autre';
      if (specName.includes('coupe') || specName.includes('coiffure')) {
        category = 'coupe';
      } else if (specName.includes('color') || specName.includes('teint')) {
        category = 'coloration';
      } else if (specName.includes('soin') || specName.includes('masque')) {
        category = 'soin';
      } else if (specName.includes('extension') || specName.includes('postiche')) {
        category = 'extension';
      }
      
      const specialty = new Specialty({
        name: specName.charAt(0).toUpperCase() + specName.slice(1),
        category,
        aliases: [specName],
        description: `Spécialité: ${specName}`,
        usageCount: 1
      });
      
      specialtyDocs.push(specialty);
    }
    
    // 4. Sauvegarder les spécialités
    const savedSpecialties = await Specialty.insertMany(specialtyDocs);
    console.log(`✅ ${savedSpecialties.length} spécialités créées`);
    
    // 5. Mettre à jour les utilisateurs
    let updatedUsers = 0;
    for (const user of users) {
      const userSpecialties = [];
      
      for (const specName of user.specialities) {
        if (specName && specName.trim()) {
          const specialty = savedSpecialties.find(s => 
            s.name.toLowerCase() === specName.trim().toLowerCase() ||
            s.aliases.includes(specName.trim().toLowerCase())
          );
          
          if (specialty) {
            userSpecialties.push({
              specialtyId: specialty._id,
              expertiseLevel: 3, // Niveau par défaut
              yearsExperience: user.experience || 0,
              certifications: [],
              isActive: true
            });
            
            // Incrémenter le compteur d'usage
            await specialty.incrementUsage();
          }
        }
      }
      
      if (userSpecialties.length > 0) {
        user.specialties = userSpecialties;
        user.specialities = undefined; // Supprimer l'ancien champ
        await user.save();
        updatedUsers++;
      }
    }
    
    console.log(`✅ ${updatedUsers} utilisateurs mis à jour`);
    console.log('🎉 Migration terminée avec succès!');
    
  } catch (error) {
    console.error('❌ Erreur lors de la migration:', error);
  } finally {
    await mongoose.disconnect();
    console.log('🔌 Déconnecté de MongoDB');
  }
};

// Exécuter la migration
migrateSpecialties();
```

### **JOUR 7-8: INTERFACE UTILISATEUR**

#### **4.1 Créer le composant `front/src/components/SpecialtyManager.tsx`**
```typescript
import React, { useState, useEffect } from 'react';
import { FaPlus, FaTimes, FaStar } from 'react-icons/fa';

interface Specialty {
  _id: string;
  name: string;
  category: string;
  expertiseLevel: number;
  yearsExperience: number;
  certifications: string[];
  isActive: boolean;
}

interface SpecialtyManagerProps {
  coiffeurId: string;
  onSpecialtiesChange: (specialties: Specialty[]) => void;
  initialSpecialties?: Specialty[];
}

export const SpecialtyManager: React.FC<SpecialtyManagerProps> = ({
  coiffeurId,
  onSpecialtiesChange,
  initialSpecialties = []
}) => {
  const [specialties, setSpecialties] = useState<Specialty[]>(initialSpecialties);
  const [availableSpecialties, setAvailableSpecialties] = useState<any[]>([]);
  const [searchQuery, setSearchQuery] = useState('');
  const [showAddForm, setShowAddForm] = useState(false);
  const [loading, setLoading] = useState(false);

  // Charger les spécialités disponibles
  useEffect(() => {
    loadAvailableSpecialties();
  }, []);

  const loadAvailableSpecialties = async () => {
    try {
      const response = await fetch('/api/specialties');
      const data = await response.json();
      setAvailableSpecialties(data);
    } catch (error) {
      console.error('Erreur chargement spécialités:', error);
    }
  };

  const searchSpecialties = async (query: string) => {
    if (query.length < 2) return;
    
    try {
      setLoading(true);
      const response = await fetch('/api/specialties/search', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ query, limit: 10 })
      });
      
      const data = await response.json();
      if (data.exact) {
        setAvailableSpecialties([data.exact]);
      } else {
        setAvailableSpecialties(data.suggestions);
      }
    } catch (error) {
      console.error('Erreur recherche spécialités:', error);
    } finally {
      setLoading(false);
    }
  };

  const addSpecialty = (specialty: any) => {
    const newSpecialty: Specialty = {
      _id: specialty._id,
      name: specialty.name,
      category: specialty.category,
      expertiseLevel: 3,
      yearsExperience: 0,
      certifications: [],
      isActive: true
    };
    
    const updatedSpecialties = [...specialties, newSpecialty];
    setSpecialties(updatedSpecialties);
    onSpecialtiesChange(updatedSpecialties);
    setSearchQuery('');
    setShowAddForm(false);
  };

  const removeSpecialty = (specialtyId: string) => {
    const updatedSpecialties = specialties.filter(s => s._id !== specialtyId);
    setSpecialties(updatedSpecialties);
    onSpecialtiesChange(updatedSpecialties);
  };

  const updateSpecialty = (specialtyId: string, field: keyof Specialty, value: any) => {
    const updatedSpecialties = specialties.map(s => 
      s._id === specialtyId ? { ...s, [field]: value } : s
    );
    setSpecialties(updatedSpecialties);
    onSpecialtiesChange(updatedSpecialties);
  };

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <h3 className="text-lg font-semibold text-gray-900">Spécialités</h3>
        <button
          onClick={() => setShowAddForm(!showAddForm)}
          className="flex items-center px-3 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors"
        >
          <FaPlus className="mr-2" />
          Ajouter une spécialité
        </button>
      </div>

      {/* Formulaire d'ajout */}
      {showAddForm && (
        <div className="p-4 border border-gray-200 rounded-lg bg-gray-50">
          <div className="mb-3">
            <label className="block text-sm font-medium text-gray-700 mb-2">
              Rechercher une spécialité
            </label>
            <input
              type="text"
              value={searchQuery}
              onChange={(e) => {
                setSearchQuery(e.target.value);
                searchSpecialties(e.target.value);
              }}
              placeholder="Tapez pour rechercher..."
              className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
            />
          </div>

          {loading && (
            <div className="text-center py-2">
              <span className="text-gray-500">Recherche en cours...</span>
            </div>
          )}

          {!loading && availableSpecialties.length > 0 && (
            <div className="space-y-2">
              {availableSpecialties.map((specialty) => (
                <div
                  key={specialty._id}
                  className="flex items-center justify-between p-3 bg-white border border-gray-200 rounded-lg hover:bg-gray-50"
                >
                  <div>
                    <span className="font-medium text-gray-900">{specialty.name}</span>
                    <span className="ml-2 text-sm text-gray-500">({specialty.category})</span>
                  </div>
                  <button
                    onClick={() => addSpecialty(specialty)}
                    className="px-3 py-1 bg-green-600 text-white text-sm rounded hover:bg-green-700 transition-colors"
                  >
                    Ajouter
                  </button>
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {/* Liste des spécialités actuelles */}
      <div className="space-y-3">
        {specialties.map((specialty) => (
          <div
            key={specialty._id}
            className="p-4 border border-gray-200 rounded-lg bg-white"
          >
            <div className="flex items-center justify-between mb-3">
              <div className="flex items-center space-x-3">
                <h4 className="font-medium text-gray-900">{specialty.name}</h4>
                <span className="px-2 py-1 text-xs bg-blue-100 text-blue-800 rounded-full">
                  {specialty.category}
                </span>
              </div>
              <button
                onClick={() => removeSpecialty(specialty._id)}
                className="text-red-600 hover:text-red-800 transition-colors"
              >
                <FaTimes />
              </button>
            </div>

            <div className="grid grid-cols-2 gap-4">
              {/* Niveau d'expertise */}
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-1">
                  Niveau d'expertise
                </label>
                <div className="flex items-center space-x-1">
                  {[1, 2, 3, 4, 5].map((level) => (
                    <button
                      key={level}
                      onClick={() => updateSpecialty(specialty._id, 'expertiseLevel', level)}
                      className={`p-1 rounded ${
                        level <= specialty.expertiseLevel
                          ? 'text-yellow-500'
                          : 'text-gray-300'
                      }`}
                    >
                      <FaStar />
                    </button>
                  ))}
                </div>
              </div>

              {/* Années d'expérience */}
              <div>
                <label className="block text-sm font-medium text-gray-700 mb-1">
                  Années d'expérience
                </label>
                <input
                  type="number"
                  min="0"
                  max="50"
                  value={specialty.yearsExperience}
                  onChange={(e) => updateSpecialty(specialty._id, 'yearsExperience', parseInt(e.target.value) || 0)}
                  className="w-full px-3 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent"
                />
              </div>
            </div>
          </div>
        ))}
      </div>

      {specialties.length === 0 && (
        <div className="text-center py-8 text-gray-500">
          Aucune spécialité configurée. Ajoutez votre première spécialité !
        </div>
      )}
    </div>
  );
};
```
