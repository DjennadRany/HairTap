```javascript
// Score = (Note × Poids) + (Disponibilité × Poids) + (Distance × Poids) + ...
const calculateScore = (coiffeur, criteria) => {
  let score = 0;
  
  // Note pondérée (0-100 points)
  score += (coiffeur.rating * 0.3) * 100;
  
  // Disponibilité (0-100 points)
  score += calculateAvailabilityScore(coiffeur, criteria.date) * 100;
  
  // Distance (0-100 points)
  score += calculateDistanceScore(coiffeur, criteria.location) * 100;
  
  // Prix (0-100 points)
  score += calculatePriceScore(coiffeur, criteria.budget) * 100;
  
  // Spécialités (0-100 points)
  score += calculateSpecialtyScore(coiffeur, criteria.service) * 100;
  
  return score;
};
```

### 2. FACTEURS DE SCORING DÉTAILLÉS

#### **A. NOTE ET RÉPUTATION (30% du score total)**
```javascript
const calculateRatingScore = (coiffeur) => {
  const baseRating = coiffeur.rating || 0;
  const reviewCount = coiffeur.reviews?.length || 0;
  const satisfactionRate = coiffeur.satisfactionRate || 0;
  
  // Formule: Note × (1 + log10(nombre_avis)) × taux_satisfaction
  return baseRating * (1 + Math.log10(Math.max(reviewCount, 1))) * satisfactionRate;
};
```

#### **B. DISPONIBILITÉ (25% du score total)**
```javascript
const calculateAvailabilityScore = (coiffeur, requestedDate) => {
  const dayOfWeek = new Date(requestedDate).getDay();
  const requestedTime = new Date(requestedDate).getHours();
  
  // Vérifier les créneaux disponibles
  const availableSlots = coiffeur.workingHours[dayOfWeek]?.slots || [];
  const matchingSlots = availableSlots.filter(slot => 
    slot.startTime <= requestedTime && 
    slot.endTime > requestedTime && 
    slot.status === 'available'
  );
  
  // Score basé sur le nombre de créneaux disponibles
  return Math.min(matchingSlots.length / 3, 1); // Max 1.0
};
```

#### **C. DISTANCE GÉOGRAPHIQUE (20% du score total)**
```javascript
const calculateDistanceScore = (coiffeur, clientLocation) => {
  const coiffeurLocation = coiffeur.addresses.salon.coordinates;
  const distance = calculateHaversineDistance(clientLocation, coiffeurLocation);
  
  // Score inversement proportionnel à la distance
  // 0km = 100 points, 50km+ = 0 points
  const maxDistance = 50; // km
  return Math.max(0, 1 - (distance / maxDistance));
};
```

#### **D. PRIX ET BUDGET (15% du score total)**
```javascript
const calculatePriceScore = (coiffeur, clientBudget) => {
  const servicePrice = coiffeur.services.find(s => s.name === clientBudget.service)?.price || 0;
  
  if (servicePrice === 0) return 0;
  
  // Score basé sur la proximité du budget
  const budgetDiff = Math.abs(servicePrice - clientBudget.amount) / clientBudget.amount;
  return Math.max(0, 1 - budgetDiff);
};
```

#### **E. SPÉCIALITÉS ET EXPERTISE (10% du score total)**
```javascript
const calculateSpecialtyScore = (coiffeur, requestedService) => {
  const matchingSpecialty = coiffeur.specialties.find(s => 
    s.name.toLowerCase().includes(requestedService.toLowerCase())
  );
  
  if (!matchingSpecialty) return 0;
  
  // Score basé sur le niveau d'expertise et l'expérience
  const expertiseLevel = matchingSpecialty.expertiseLevel || 1; // 1-5
  const yearsExperience = matchingSpecialty.yearsExperience || 0;
  
  return (expertiseLevel / 5) * 0.7 + Math.min(yearsExperience / 10, 1) * 0.3;
};
```

## 🔄 ALGORITHME DE RECHERCHE COMPLET

### 1. FLUX DE RECHERCHE
```javascript
// 1. Récupération des critères client
const searchCriteria = {
  service: "coupe_homme",
  location: { lat: 48.8566, lng: 2.3522 },
  date: "2025-01-20T14:00:00Z",
  budget: { min: 20, max: 80 },
  workingMode: "salon", // ou "domicile" ou "both"
  maxDistance: 25, // km
  minRating: 4.0
};

// 2. Filtrage préliminaire (MongoDB)
const preliminaryResults = await User.find({
  role: 'coiffeur',
  status: 'active',
  isOnline: true,
  'addresses.salon.coordinates': {
    $near: {
      $geometry: {
        type: 'Point',
        coordinates: [searchCriteria.location.lng, searchCriteria.location.lat]
      },
      $maxDistance: searchCriteria.maxDistance * 1000 // mètres
    }
  },
  'workingMode': { $in: [searchCriteria.workingMode, 'both'] },
  'rating': { $gte: searchCriteria.minRating }
}).limit(100);

// 3. Scoring détaillé
const scoredResults = preliminaryResults.map(coiffeur => ({
  coiffeur,
  score: calculateScore(coiffeur, searchCriteria),
  details: {
    rating: calculateRatingScore(coiffeur),
    availability: calculateAvailabilityScore(coiffeur, searchCriteria.date),
    distance: calculateDistanceScore(coiffeur, searchCriteria.location),
    price: calculatePriceScore(coiffeur, searchCriteria.budget),
    specialty: calculateSpecialtyScore(coiffeur, searchCriteria.service)
  }
}));

// 4. Tri par score et pagination
const sortedResults = scoredResults
  .sort((a, b) => b.score - a.score)
  .slice(0, 20); // Top 20 résultats

// 5. Cache des résultats
await redis.setex(cacheKey, 300, JSON.stringify(sortedResults)); // 5 minutes
```

## 🔒 GESTION DES RÉSERVATIONS ET CONFLITS

### 1. VÉRIFICATION DES CRÉNEAUX
```javascript
const checkSlotAvailability = async (coiffeurId, date, service) => {
  const slot = await WorkingSlot.findOne({
    coiffeurId,
    date,
    startTime: { $lte: service.startTime },
    endTime: { $gt: service.endTime },
    status: 'available'
  });
  
  if (!slot) return false;
  
  // Vérifier qu'il n'y a pas de réservation en conflit
  const conflictingBooking = await Booking.findOne({
    coiffeurId,
    date: { $gte: service.startTime, $lt: service.endTime },
    status: { $in: ['confirmed', 'pending'] }
  });
  
  return !conflictingBooking;
};
```

### 2. SYSTÈME DE RÉSERVATION ATOMIQUE
```javascript
const bookSlot = async (bookingData) => {
  const session = await mongoose.startSession();
  
  try {
    await session.withTransaction(async () => {
      // 1. Vérifier la disponibilité
      const isAvailable = await checkSlotAvailability(/* ... */);
      if (!isAvailable) throw new Error('Créneau non disponible');
      
      // 2. Créer la réservation
      const booking = new Booking(bookingData);
      await booking.save({ session });
      
      // 3. Mettre à jour le créneau
      await WorkingSlot.updateOne(

      );
```
