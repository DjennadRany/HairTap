```javascript
// back/scripts/migrate-working-slots.js
const migrateWorkingSlots = async () => {
  const coiffeurs = await User.find({ role: 'coiffeur' });
  
  for (const coiffeur of coiffeurs) {
    // Créer des créneaux par défaut (Lundi-Vendredi, 9h-18h)
    const defaultSlots = [];
    
    for (let day = 1; day <= 5; day++) { // Lundi = 1, Vendredi = 5
      defaultSlots.push({
        coiffeurId: coiffeur._id,
        dayOfWeek: day,
        startTime: 9,
        endTime: 18,
        serviceTypes: ['coupe', 'coloration', 'soin'],
        availableAt: coiffeur.workingMode === 'both' ? 'both' : 
                    coiffeur.workingMode === 'domicile' ? 'domicile' : 'salon',
        status: 'available',
        maxBookings: 1
      });
    }
    
    // Ajouter le samedi si le coiffeur travaille le weekend
    if (coiffeur.workingMode !== 'salon') {
      defaultSlots.push({
        coiffeurId: coiffeur._id,
        dayOfWeek: 6, // Samedi
        startTime: 9,
        endTime: 17,
        serviceTypes: ['coupe', 'coloration', 'soin'],
        availableAt: coiffeur.workingMode === 'both' ? 'both' : 
                    coiffeur.workingMode === 'domicile' ? 'domicile' : 'salon',
        status: 'available',
        maxBookings: 1
      });
    }
    
    for (const slot of defaultSlots) {
      const workingSlot = new WorkingSlot(slot);
      await workingSlot.save();
    }
    
    console.log(`✅ Créneaux migrés pour ${coiffeur.name}`);
  }
};
```

```javascript
  try {
    const { date, serviceType } = req.query;
    let query = { coiffeurId: req.params.coiffeurId };

      const dayOfWeek = new Date(date).getDay();
```

```javascript
// Validation côté serveur
const validateSpecialty = (specialty) => {
  const errors = [];
  
  if (!specialty.name || specialty.name.trim().length < 2) {
    errors.push('Le nom de la spécialité doit contenir au moins 2 caractères');
  }
  
  if (specialty.expertiseLevel < 1 || specialty.expertiseLevel > 5) {
    errors.push('Le niveau d\'expertise doit être entre 1 et 5');
  }
  
  if (specialty.yearsExperience < 0) {
    errors.push('Les années d\'expérience ne peuvent pas être négatives');
  }
  
  return errors;
};
```
